print("\nRELATIONAL OPERATORS")
print("a < b =", a < b)
print("a <= b =", a <= b)
print("a > b =", a > b)
print("a >= b =", a >= b)
print("a == b =", a == b)
print("a != b =", a != b)

print("\nLOGICAL OPERATORS")
print("a < b and a > 5 =", a < b and a > 5)
print("a < b or a < 5 =", a < b or a < 5)
print("not(a < b) =", not(a < b))

print("\nBITWISE OPERATORS")
print("a & b =", a & b)
print("a | b =", a | b)
print("a ^ b =", a ^ b)
print("~a =", ~a)
print("a << 1 =", a << 1)
print("a >> 1 =", a >> 1)
